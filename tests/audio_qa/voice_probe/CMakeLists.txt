cmake_minimum_required(VERSION 3.25)
project(AytherVoiceProbe LANGUAGES CXX)
enable_testing()
if(NOT WIN32)
    message(FATAL_ERROR "This experiment pins the Windows reference")
endif()
set(reference_header "${QA_ENGINE_REFERENCE_SOURCE}/include/ayther/audio_hd_mixer.h")
file(SHA256 "${reference_header}" source_hash)
if(NOT source_hash STREQUAL "f47f24ca73632799da944f2ec5b8425221c2deda475291f1c9263fd02448dae8")
    message(FATAL_ERROR "Mixer reference changed; review the observation anchors")
endif()
file(READ "${reference_header}" mixer)
string(REPLACE "\r\n" "\n" mixer "${mixer}")
function(observe anchor replacement)
    string(FIND "${mixer}" "${anchor}" position)
    if(position LESS 0)
        message(FATAL_ERROR "Observation anchor missing: ${anchor}")
    endif()
    string(REPLACE "${anchor}" "${replacement}" mixer "${mixer}")
    set(mixer "${mixer}" PARENT_SCOPE)
endfunction()
observe("    struct Voice {" "    struct Voice {\n        uint64_t qa_occurrence = 0; // Test-only identity; never read by mixing decisions.")
observe("        voices_.push_back(std::move(v));" "        v.qa_occurrence = started_ + 1;\n        voices_.push_back(std::move(v));")
observe("                if (v.pos >= cycle_end) {" "                const size_t qa_cursor_before = v.pos;\n                if (v.pos >= cycle_end) {")
observe("                const size_t o = f * 2;" "                const size_t o = f * 2;\n                const int16_t qa_before_left = out[o];\n                const int16_t qa_before_right = out[o + 1];")
observe("                ++v.pos;" "                ++v.pos;\n                qa_record({v.qa_occurrence, v.key, block_start + f, qa_cursor_before,\n                           s / 2, v.pos, g, v.fade_left, pcm[s], pcm[s + 1],\n                           qa_before_left, qa_before_right, out[o], out[o + 1]});")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/observed-header")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/observed-header/audio_hd_mixer.h"
    "// AYTHER QA disposable observation copy; original mixing decisions are unchanged.\n${mixer}")
foreach(kind IN ITEMS reference observed)
    add_executable(audio_qa_voice_${kind} voice_probe.cpp)
    target_compile_features(audio_qa_voice_${kind} PRIVATE cxx_std_20)
    target_compile_options(audio_qa_voice_${kind} PRIVATE /W4 /WX /permissive- /EHsc)
    set_target_properties(audio_qa_voice_${kind} PROPERTIES CXX_EXTENSIONS OFF)
    if(kind STREQUAL "observed")
        target_compile_definitions(audio_qa_voice_${kind} PRIVATE QA_EXPECT_OBSERVER=1)
        target_include_directories(audio_qa_voice_${kind} SYSTEM PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/observed-header")
    else()
        target_compile_definitions(audio_qa_voice_${kind} PRIVATE QA_EXPECT_OBSERVER=0)
        target_include_directories(audio_qa_voice_${kind} SYSTEM PRIVATE "${QA_ENGINE_REFERENCE_SOURCE}/include/ayther")
    endif()
endforeach()
foreach(mode IN ITEMS complete lost)
    add_test(NAME audio_qa_voice_${mode}
        COMMAND ${CMAKE_COMMAND} "-DREFERENCE=$<TARGET_FILE:audio_qa_voice_reference>"
            "-DOBSERVED=$<TARGET_FILE:audio_qa_voice_observed>" "-DMODE=${mode}"
            "-DOUTPUT_ROOT=${CMAKE_CURRENT_BINARY_DIR}/results/$<CONFIG>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/verify.cmake")
    set_tests_properties(audio_qa_voice_${mode} PROPERTIES TIMEOUT 10 LABELS "audio_qa;integration")
endforeach()
